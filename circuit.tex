% ============================================================
%  NAND gate - transistor-level schematic (COMPLEMENTARY / CMOS-style)
%  Pull-up : two 2N3906 (PNP) in PARALLEL  (emitters -> +5 V)
%  Pull-down: two 2N3904 (NPN) in SERIES   (bottom emitter -> GND)
%  Each input drives one PNP and one NPN through its OWN base resistor.
%    Output LOW only when BOTH inputs HIGH (both NPNs conduct);
%    otherwise a PNP pulls the output HIGH.   => Y = NOT(A.B)
%  Compile:  pdflatex circuit.tex
% ============================================================
\documentclass[border=18pt]{standalone}
\usepackage{circuitikz}
\usepackage{amsmath}
\usetikzlibrary{calc}

\newcommand{\pinlabelsN}[1]{%
  \node[red,font=\bfseries\footnotesize,anchor=west] at ($(#1.C)+(0.22,0.16)$){C};
  \node[red,font=\bfseries\footnotesize,anchor=south east] at ($(#1.B)+(-0.08,0.14)$){B};
  \node[red,font=\bfseries\footnotesize,anchor=west] at ($(#1.E)+(0.22,-0.16)$){E};
}
\newcommand{\pinlabelsP}[1]{%
  \node[red,font=\bfseries\footnotesize,anchor=west] at ($(#1.E)+(0.22,0.16)$){E};
  \node[red,font=\bfseries\footnotesize,anchor=north east] at ($(#1.B)+(-0.08,-0.14)$){B};
  \node[red,font=\bfseries\footnotesize,anchor=west] at ($(#1.C)+(0.22,-0.16)$){C};
}
\tikzset{gnd/.pic={
  \draw (0,0) -- (0,-0.18) (-0.34,-0.18) -- (0.34,-0.18)
        (-0.21,-0.37) -- (0.21,-0.37) (-0.08,-0.56) -- (0.08,-0.56);
}}

\begin{document}
\begin{circuitikz}[american, line width=1.1pt]
  \ctikzset{tripoles/thickness=1, bipoles/thickness=1}

  % ---------- supply rail (+5 V) ----------
  \draw (-5,12) -- (21,12);
  \node[anchor=south] at (8,12.05) {\textbf{$V_{CC}=+5\,\mathrm{V}$}};

  % ===================== PULL-UP : two PNP in parallel =====================
  \node[pnp] (pa) at (10,9.4) {};
  \node[pnp] (pb) at (15,9.4) {};
  \pinlabelsP{pa}\pinlabelsP{pb}
  \node[font=\footnotesize,align=left,anchor=west] at (10.95,9.4) {Q3\\2N3906};
  \node[font=\footnotesize,align=left,anchor=west] at (15.95,9.4) {Q4\\2N3906};
  \draw (pa.E) -- (10,12);  \fill (10,12) circle (1.6pt);
  \draw (pb.E) -- (15,12);  \fill (15,12) circle (1.6pt);
  % both collectors -> output bus
  \draw (pa.C) -- (10,6.4);
  \draw (pb.C) -- (15,6.4);
  \draw (10,6.4) -- (15,6.4);

  % ===================== OUTPUT node =====================
  \coordinate (OUT) at (12.5,6.4);
  \fill (OUT) circle (1.6pt);
  \draw (OUT) -- (20.5,6.4) node[ocirc]{} node[right]{Output $Y=\overline{A\cdot B}$};

  % ===================== PULL-DOWN : two NPN in series =====================
  \node[npn] (na) at (12.5,4.4) {};
  \node[npn] (nb) at (12.5,1.6) {};
  \pinlabelsN{na}\pinlabelsN{nb}
  \node[font=\footnotesize,align=left,anchor=west] at (13.45,4.4) {Q1\\2N3904};
  \node[font=\footnotesize,align=left,anchor=west] at (13.45,1.6) {Q2\\2N3904};
  \draw (na.C) -- (OUT);
  \draw (na.E) -- (nb.C);
  \draw (nb.E) -- (12.5,0.4) pic{gnd};

  % ===================== INPUT A (drives PNP Q3 and NPN Q1) =====================
  \draw (-5,7.2) node[ocirc]{} node[left]{Input $A$} -- (2.5,7.2);
  \draw (2.5,9.4) -- (2.5,4.4);          % A rail
  \fill (2.5,7.2) circle (1.6pt); \fill (2.5,9.4) circle (1.6pt); \fill (2.5,4.4) circle (1.6pt);
  % A -> R_A1 -> PNP Q3 base  (plain wire clears the B rail, then the resistor)
  \draw (2.5,9.4) -- (5.5,9.4) to[R, l={$R_{A1}=10\,\mathrm{k}\Omega$}] (pa.B);
  % A -> R_A2 -> NPN Q1 base
  \draw (2.5,4.4) -- (6.5,4.4) to[R, l={$R_{A2}=10\,\mathrm{k}\Omega$}] (na.B);
  % input LED on A
  \draw (2.5,4.4) -- (2.5,3.0);
  \draw (2.5,3.0) to[R, l_={$R_{inA}=220\,\Omega$}] (2.5,1.4)
        to[leD, l_={LED$_{A}$}] (2.5,-0.2) -- (2.5,-0.25) pic{gnd};

  % ===================== INPUT B (drives PNP Q4 and NPN Q2) =====================
  \draw (-5,5.6) node[ocirc]{} node[left]{Input $B$} -- (4.5,5.6);
  \draw (4.5,10.9) -- (4.5,1.6);         % B rail
  \fill (4.5,5.6) circle (1.6pt); \fill (4.5,10.9) circle (1.6pt); \fill (4.5,1.6) circle (1.6pt);
  % B -> R_B1 -> PNP Q4 base  (over the top, straight down into the base)
  \draw (4.5,10.9) -- (8.0,10.9) to[R, l={$R_{B1}=10\,\mathrm{k}\Omega$}] ($(pb.B)+(0,1.5)$) -- (pb.B);
  % B -> R_B2 -> NPN Q2 base
  \draw (4.5,1.6) -- (6.5,1.6) to[R, l_={$R_{B2}=10\,\mathrm{k}\Omega$}] (nb.B);
  % input LED on B
  \draw (4.5,1.6) -- (4.5,0.3);
  \draw (4.5,0.3) to[R, l={$R_{inB}=220\,\Omega$}] (4.5,-1.3)
        to[leD, l={LED$_{B}$}] (4.5,-2.9) -- (4.5,-2.95) pic{gnd};

  % ===================== OUTPUT LED =====================
  \fill (17.5,6.4) circle (1.6pt);
  \draw (17.5,6.4) to[R, l={$R_{out}=220\,\Omega$}] (17.5,4.5)
        to[leD, l={LED$_{\mathrm{out}}$}] (17.5,2.9) -- (17.5,2.85) pic{gnd};

\end{circuitikz}
\end{document}
