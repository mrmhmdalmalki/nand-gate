% ============================================================
%  NAND gate - transistor-level realization
%  Two NPN transistors in SERIES (stacked) with one pull-up.
%    Output is pulled to ground ONLY when BOTH conduct (A=B=1).
%    If either input is 0 -> the chain breaks -> output to +5 V.
%  => Y = NOT(A AND B)
%  Compile:  pdflatex circuit.tex
% ============================================================
\documentclass[border=12pt]{standalone}
\usepackage{circuitikz}
\usepackage{amsmath}

\begin{document}
\begin{circuitikz}[american, line width=0.7pt]

  % ---------- Supply rail (+5 V) ----------
  \draw (-1,6.5) -- (7,6.5);
  \node[anchor=south] at (3,6.55) {\textbf{$V_{CC} = +5\,\mathrm{V}$}};
  \fill (3,6.5) circle (1.6pt);

  % ---------- transistors (in series / stacked) ----------
  \node[npn] (q1) at (3,4) {};   % top transistor (input A)
  \node[npn] (q2) at (3,1.3) {}; % bottom transistor (input B)
  \node[anchor=west] at ($(q1.E)+(0.45,0.15)$) {Q1};
  \node[anchor=west] at ($(q2.E)+(0.45,0.15)$) {Q2};

  % series link: emitter of Q1 -> collector of Q2
  \draw (q1.E) -- (q2.C);

  % bottom emitter -> ground
  \draw (q2.E) -- ++(0,-1.0) node[ground]{};

  % pull-up from rail to top collector (= output node)
  \draw (q1.C) to[R, l_={$R_C=1\,\mathrm{k}\Omega$}] (3,6.5);

  % inputs
  \draw (-2,4) node[ocirc]{} node[left]{Input $A$}
        to[R, l={$R_{B1}=10\,\mathrm{k}\Omega$}] (q1.B);
  \draw (-2,1.3) node[ocirc]{} node[left]{Input $B$}
        to[R, l={$R_{B2}=10\,\mathrm{k}\Omega$}] (q2.B);

  % output at the top collector
  \draw (q1.C) -- ++(3,0) node[ocirc]{} node[right]{Output $Y=\overline{A\cdot B}$};

\end{circuitikz}
\end{document}
